<script setup lang="ts">
// 已选定的两种外观通过设置切换；业务入口仍为预览交互。
import { onMounted, onUnmounted, ref, watch } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import type { LocationQueryRaw } from 'vue-router'
import AppIcon from '../../components/AppIcon.vue'
import VariantD from './VariantD.vue'
import './prototype-light-palettes.css'

const route = useRoute()
const router = useRouter()
type Appearance = 'D' | 'F'
const preferenceKey = 'tuanzi-ui-appearance'
// 明确的预览链接优先；普通入口恢复本地外观，不改旧版页面的主题存储。
function initialAppearance(): Appearance {
  if (route.query.variant === 'D' || route.query.variant === 'F') return route.query.variant
  try { return localStorage.getItem(preferenceKey) === 'F' ? 'F' : 'D' } catch { return 'D' }
}
const variant = ref<Appearance>(initialAppearance())
const settingsOpen = ref(false)
const labels = { D: '深色苔绿', F: '米白陶土' }
const storageNotice = ref('')
const dialog = ref<HTMLDialogElement | null>(null)
const previewTitle = ref('')
const originalTitle = document.title
const examples = ['把一篇长文整理成清晰的摘要', '围绕一个主题收集灵感和参考', '将一个初步想法发展成创作方案']

function select(key: Appearance) {
  variant.value = key
  try {
    localStorage.setItem(preferenceKey, key)
    storageNotice.value = '外观已保存，下次打开自动使用。'
  } catch {
    storageNotice.value = '已切换外观；当前浏览器无法保存，下次打开将使用默认外观。'
  }
}
function preview(title: string) {
  settingsOpen.value = title === '设置'
  previewTitle.value = title
  dialog.value?.showModal()
}
watch(variant, () => {
  document.title = `团子 AI · ${labels[variant.value]} · UI 原型`
}, { immediate: true })
onMounted(() => {
  document.documentElement.classList.add('viewing-ui-prototype')
  // 将一次性的配色直达链接归一为设置入口，刷新时以已保存的用户偏好为准。
  if (route.query.variant) {
    select(variant.value)
    const query: LocationQueryRaw = { ...route.query, prototype: '1' }
    delete query.variant
    router.replace({ query })
  }
})
onUnmounted(() => {
  document.title = originalTitle
  document.documentElement.classList.remove('viewing-ui-prototype')
})
</script>

<template>
  <div
    class="ui-prototype"
    :class="`prototype-${variant.toLowerCase()}`"
  >
    <VariantD
      @preview="preview"
    />
    <dialog
      ref="dialog"
      class="prototype-dialog"
      aria-labelledby="prototype-dialog-title"
      @click="event => { if (event.target === dialog) dialog?.close() }"
    >
      <div class="dialog-top">
        <span>{{ settingsOpen ? '设置 / 外观' : '交互预览' }}</span><button
          aria-label="关闭预览"
          @click="dialog?.close()"
        >
          <AppIcon
            name="x"
            :size="20"
          />
        </button>
      </div>
      <template v-if="settingsOpen">
        <h2 id="prototype-dialog-title">
          选择你喜欢的外观
        </h2>
        <p>为你的 AI 工作空间，选一个舒服的颜色。</p>
        <div
          class="appearance-options"
          role="group"
          aria-label="外观方案"
        >
          <button
            v-for="key in (['D', 'F'] as const)"
            :key="key"
            class="appearance-option"
            :class="{ selected: variant === key }"
            :aria-pressed="variant === key"
            @click="select(key)"
          >
            <span
              class="appearance-sample"
              :class="key === 'D' ? 'sample-dark' : 'sample-light'"
              aria-hidden="true"
            ><i /><span><b /><em /><em /></span></span>
            <span class="appearance-label">{{ labels[key] }}<AppIcon
              v-if="variant === key"
              name="check"
              :size="17"
            /></span>
            <small>{{ key === 'D' ? '炭黑底色 · 柔和苔绿' : '暖米白底色 · 陶土橙' }}</small>
          </button>
        </div>
        <p
          class="appearance-notice"
          role="status"
        >
          {{ storageNotice || '选择后立即生效，并保存在此浏览器。' }}
        </p>
        <button
          class="dialog-return"
          @click="dialog?.close()"
        >
          完成
        </button>
      </template>
      <template v-else>
        <span class="dialog-icon"><AppIcon
          name="sparkles"
          :size="28"
        /></span>
        <h2 id="prototype-dialog-title">
          {{ previewTitle }}
        </h2>
        <p>这里展示所选入口的交互反馈。当前是视觉原型，未发送 AI 请求。</p>
        <div class="dialog-examples">
          <span
            v-for="example in examples"
            :key="example"
          ><AppIcon
            name="check"
            :size="16"
          />{{ example }}</span>
        </div>
        <button
          class="dialog-return"
          @click="dialog?.close()"
        >
          继续比较页面
        </button>
      </template>
    </dialog>
  </div>
</template>
